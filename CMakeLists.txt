cmake_minimum_required(VERSION 3.16)
project(TKApp VERSION 2.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
option(TK_BUILD_TESTS "Build local tests (no connection to the production SIP server)" ON)
set(TK_HOST_MOC "" CACHE FILEPATH "Host moc for a MinGW cross-build")
set(TK_HOST_RCC "" CACHE FILEPATH "Host rcc for a MinGW cross-build")
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Gui Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Widgets)
if(CMAKE_CROSSCOMPILING AND TK_HOST_MOC)
    set_property(TARGET Qt${QT_VERSION_MAJOR}::moc PROPERTY IMPORTED_LOCATION "${TK_HOST_MOC}")
    if(TK_HOST_RCC)
        set_property(TARGET Qt${QT_VERSION_MAJOR}::rcc PROPERTY IMPORTED_LOCATION "${TK_HOST_RCC}")
    endif()
endif()
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(PJ REQUIRED libpjproject>=2.15.1)
pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
pkg_check_modules(OGG REQUIRED IMPORTED_TARGET ogg)
# We use the C PJSUA API, not PJSUA2 (some MinGW installations omit that archive).
set(PJ_LINK_LIBS ${PJ_STATIC_LIBRARIES})
list(FILTER PJ_LINK_LIBS EXCLUDE REGEX "^pjsua2")
add_library(tk_audio STATIC src/AudioRecorder.cpp src/AudioRecorder.h src/AppSettings.cpp src/AppSettings.h)
target_include_directories(tk_audio PUBLIC src)
target_link_libraries(tk_audio PUBLIC Qt${QT_VERSION_MAJOR}::Core PkgConfig::OPUS PkgConfig::OGG Threads::Threads)
add_library(tk_sip STATIC src/SipEngine.cpp src/SipEngine.h)
target_include_directories(tk_sip SYSTEM PUBLIC ${PJ_INCLUDE_DIRS})
target_compile_options(tk_sip PUBLIC ${PJ_CFLAGS_OTHER})
target_link_directories(tk_sip PUBLIC ${PJ_STATIC_LIBRARY_DIRS} ${OPUS_LIBRARY_DIRS} ${OGG_LIBRARY_DIRS})
target_link_libraries(tk_sip PUBLIC tk_audio ${PJ_LINK_LIBS})
if(WIN32)
    target_link_libraries(tk_sip PUBLIC winmm ole32 ws2_32 wsock32 iphlpapi)
endif()
add_executable(tk-app WIN32 src/main.cpp src/MainWindow.cpp src/MainWindow.h src/SettingsDialog.cpp src/SettingsDialog.h src/Icons.cpp src/Icons.h src/BrandWidgets.cpp src/BrandWidgets.h packaging/resources.qrc)
if(WIN32)
    target_sources(tk-app PRIVATE packaging/windows.rc)
endif()
target_link_libraries(tk-app PRIVATE tk_sip Qt${QT_VERSION_MAJOR}::Widgets)
target_compile_definitions(tk-app PRIVATE TK_APP_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(tk-app PRIVATE /utf-8 /W4)
else()
    foreach(t tk_audio tk_sip tk-app)
        target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
    endforeach()
endif()
install(TARGETS tk-app RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/tk-app.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/tk-app.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE README.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/tk-app)
if(BUILD_TESTING AND TK_BUILD_TESTS)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(tk-unit-tests tests/UnitTests.cpp)
    target_link_libraries(tk-unit-tests PRIVATE tk_audio Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME unit COMMAND tk-unit-tests)
    add_executable(tk-sip-tests tests/SipTests.cpp)
    target_link_libraries(tk-sip-tests PRIVATE tk_sip Qt${QT_VERSION_MAJOR}::Test)
    target_compile_definitions(tk-sip-tests PRIVATE TK_TEST_PEER="${CMAKE_CURRENT_SOURCE_DIR}/tests/sip_peer.py")
    add_test(NAME sip-loopback COMMAND tk-sip-tests)
    set_tests_properties(sip-loopback PROPERTIES TIMEOUT 60)
    add_executable(tk-ui-tests tests/UiTests.cpp src/MainWindow.cpp src/SettingsDialog.cpp src/Icons.cpp src/BrandWidgets.cpp packaging/resources.qrc)
    target_link_libraries(tk-ui-tests PRIVATE tk_sip Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME interface COMMAND tk-ui-tests)
    set_tests_properties(interface PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
